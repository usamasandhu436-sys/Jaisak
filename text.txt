/**
 * Renders personality cards into the target container element.
 * @param {string[]} names - Array of names to render.
 * @param {string} target - ID of the container element.
 */
function addPeople(names, target) {
  const container = document.getElementById(target);
  if (!container) return;

  names.forEach(name => {
    const article = document.createElement("article");
    article.className = "person";

    const heading = document.createElement("h3");
    heading.textContent = name;

    const paragraph = document.createElement("p");
    paragraph.textContent =
      "Listed in the Jaisak community directory. Biography and achievements to be confirmed.";

    article.appendChild(heading);
    article.appendChild(paragraph);
    container.appendChild(article);
  });
}
